Estructura para Supabase

Esquema de base de datos PostgreSQL con 5 tablas, seguridad RLS y configuración de almacenamiento de fotos.

1 Crear Proyecto

Entra a supabase.com y crea tu proyecto gratis.

2 Abrir SQL Editor

En el menú izquierdo de Supabase, entra a SQL Editor > New Query.

3 Pegar & Ejecutar

Pega el código SQL de abajo y presiona el botón verde RUN.

4 Crear Buckets

En Storage, crea 2 carpetas públicas: covers y photos.

supabase_schema.sql PostgreSQL 15 
-- ==============================================================================
-- STUDIO NVF - NICOLE VALERIO FOTOGRAFÍA EDITORIAL
-- Arquitectura de Base de Datos para Supabase (PostgreSQL + RLS + Storage)
-- ==============================================================================

-- 1. EXTENSIÓN UUID
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 2. TABLA: books (Catálogo de Colecciones)
CREATE TABLE public.books (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    slug TEXT UNIQUE NOT NULL,                          -- ej: 'familia', 'boda'
    badge_number TEXT NOT NULL,                         -- ej: '01', '02'
    title TEXT NOT NULL,                                -- ej: 'Familia'
    subtitle TEXT,
    category TEXT DEFAULT 'EDITORIAL',
    cover_url TEXT NOT NULL,
    photo_count INTEGER DEFAULT 0,
    sort_order INTEGER DEFAULT 0,
    is_published BOOLEAN DEFAULT true,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. TABLA: photos (Galería en Alta Resolución)
CREATE TABLE public.photos (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    book_id UUID NOT NULL REFERENCES public.books(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    caption TEXT,
    url_large TEXT NOT NULL,
    url_thumb TEXT,
    width INTEGER,
    height INTEGER,
    is_cover BOOLEAN DEFAULT false,
    sort_order INTEGER DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 4. TABLA: bookings (Citas y Reservas del Calendario)
CREATE TABLE public.bookings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    client_name TEXT NOT NULL,
    client_email TEXT NOT NULL,
    client_phone TEXT NOT NULL,
    session_type TEXT NOT NULL,                         -- 'Familia', 'Boda', 'Embarazo', etc.
    booking_date DATE NOT NULL,                         -- Fecha (AAAA-MM-DD)
    booking_time TIME NOT NULL DEFAULT '10:30:00',      -- Hora (HH:MM)
    location TEXT,
    message TEXT,
    status TEXT DEFAULT 'Pendiente' CHECK (
        status IN ('Pendiente', 'Confirmada', 'Completada', 'Cancelada')
    ),
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 5. TABLA: reviews (Calificaciones y Reseñas)
CREATE TABLE public.reviews (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    author_name TEXT NOT NULL,
    rating INTEGER NOT NULL CHECK (rating >= 1 AND rating <= 5),
    session_type TEXT,
    review_text TEXT NOT NULL,
    is_approved BOOLEAN DEFAULT true,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 6. POLÍTICAS DE SEGURIDAD (ROW LEVEL SECURITY - RLS)
ALTER TABLE public.books ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.photos ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.bookings ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Público puede ver books" ON public.books FOR SELECT USING (is_published = true);
CREATE POLICY "Admin gestiona books" ON public.books FOR ALL TO authenticated USING (true);

CREATE POLICY "Público puede ver fotos" ON public.photos FOR SELECT USING (true);
CREATE POLICY "Admin gestiona fotos" ON public.photos FOR ALL TO authenticated USING (true);

CREATE POLICY "Clientes pueden enviar reservas" ON public.bookings FOR INSERT TO anon, authenticated WITH CHECK (true);
CREATE POLICY "Admin gestiona reservas" ON public.bookings FOR ALL TO authenticated USING (true);

-- 7. STORAGE BUCKETS (En Supabase Storage crea los buckets públicos: 'covers' y 'photos')
INSERT INTO storage.buckets (id, name, public)
VALUES ('covers', 'covers', true), ('photos', 'photos', true)
ON CONFLICT (id) DO NOTHING;
